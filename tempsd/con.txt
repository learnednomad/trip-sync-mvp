  6. Identify Potential Risks and Unforeseen Issues

  Technology Risk Assessment

  High Risk Issues:

  1. React 19.0.0 Bleeding Edge Risk 🚨
    - Issue: Too new, limited community solutions
    - Impact: Debugging difficulties, lack of Stack Overflow answers
    - Mitigation: Consider downgrading to React 18.3.x for stability
  2. Offline-First Complexity ⚠️
    - Issue: Zustand + TanStack Query + MMKV synchronization
    - Unforeseen: Conflict resolution between offline edits
    - Mitigation: Implement CRDT-based sync or last-write-wins with user choice
  3. Expo Ejection Risk ⚠️
    - Issue: May need native modules Expo doesn't support
    - Examples: Advanced camera features, custom Bluetooth protocols
    - Mitigation: Plan ejection strategy, use Expo development builds
  4. NativeWind Performance ⚠️
    - Issue: Runtime style computation overhead
    - Impact: Slower on low-end Android devices
    - Mitigation: Profile early, consider StyleSheet for hot paths
  5. Multi-Currency Calculation Precision 🚨
    - Issue: JavaScript floating-point errors in financial calculations
    - Impact: Incorrect expense splitting
    - Mitigation: Use decimal.js or big.js for money math

  Edge Cases Not Considered:
  - RTL language support with NativeWind
  - Accessibility with custom components
  - Deep linking with Expo Router in production
  - Background sync while app is killed

  - services/api/: External API communication

  - services/storage/: Local persistence layer

  4. State Management Structure
  - Separate stores by domain (auth, trip, ui)
  - Offline store for queue management

  - Co-located unit tests in __tests__/
  - Maestro flows in .maestro/ for E2E
  - MSW mocks for API testing

  Trade-offs Made:
  - File-based routing over config: Better DX, automatic code splitting
  - Feature folders in components: Easier to find related components
  - Separate services layer: Clear business logic separation
  - Multiple stores over single: Better code splitting and clarity

  // components/ui/LoadingButton.tsx
  import { Button, ButtonProps } from '@/components/primitives/button'
  import { ActivityIndicator } from 'react-native'

  interface LoadingButtonProps extends ButtonProps {
    loading?: boolean
  }

  export function LoadingButton({
    loading,
    children,
    disabled,
    ...props
  }: LoadingButtonProps) {
    return (
      <Button disabled={disabled || loading} {...props}>
        {loading ? <ActivityIndicator /> : children}
      </Button>
    )
  }

  export function cn(...inputs: ClassValue[]) {

  // hooks/useOptimizedTheme.ts
  import { useMemo, useCallback } from 'react'
  import { useColorScheme } from 'nativewind'
  import { useMMKVObject } from 'react-native-mmkv'

  interface ThemeCache {
    colorScheme: 'light' | 'dark'
    customColors: Record<string, string>
    timestamp: number
  }

  export function useOptimizedTheme() {
    const { colorScheme } = useColorScheme()
    const [themeCache, setThemeCache] = useMMKVObject<ThemeCache>('theme-cache')

    // Memoize expensive theme calculations
    const computedTheme = useMemo(() => {
      // Only recompute if colorScheme changes
      if (themeCache?.colorScheme === colorScheme &&
          Date.now() - themeCache.timestamp < 60000) { // 1 min cache
        return themeCache.customColors
      }

      // Expensive theme computation
      const newTheme = {
        primary: colorScheme === 'dark' ? '#63B3ED' : '#2B6CB0',
        secondary: colorScheme === 'dark' ? '#4FD1C5' : '#2C7A7B',
        // ... more colors
      }

      // Cache the result
      setThemeCache({
        colorScheme,
        customColors: newTheme,
        timestamp: Date.now(),
      })

      return newTheme
    }, [colorScheme, themeCache])

    return computedTheme
  }

  4. Runtime Performance Monitoring

  Theme Performance Metrics

  // utils/performanceMonitor.ts
  import { PerformanceObserver, performance } from 'react-native-performance'

  export class ThemePerformanceMonitor {
    private measurements: Map<string, number[]> = new Map()

    startMeasure(name: string) {
      performance.mark(`${name}-start`)
    }

    endMeasure(name: string) {
      performance.mark(`${name}-end`)
      performance.measure(name, `${name}-start`, `${name}-end`)

      const entry = performance.getEntriesByName(name)[0]
      if (entry) {
        const current = this.measurements.get(name) || []
        current.push(entry.duration)
        this.measurements.set(name, current)

        // Alert if theme switch takes too long
        if (name === 'theme-switch' && entry.duration > 100) {
          console.warn(`Theme switch took ${entry.duration}ms`)
        }
      }
    }

    getAverageTime(name: string): number {
      const times = this.measurements.get(name) || []
      return times.reduce((a, b) => a + b, 0) / times.length
    }

    reportMetrics() {
      const report = {}
      this.measurements.forEach((times, name) => {
        report[name] = {
          average: this.getAverageTime(name),
          min: Math.min(...times),
          max: Math.max(...times),
          count: times.length,
        }
      })
      return report
    }
  }

  // Usage in theme hook
  export function useMonitoredTheme() {
    const monitor = useRef(new ThemePerformanceMonitor())
    const { colorScheme, setColorScheme } = useColorScheme()

    const switchTheme = useCallback((newScheme: 'light' | 'dark') => {
      monitor.current.startMeasure('theme-switch')
      setColorScheme(newScheme)
      monitor.current.endMeasure('theme-switch')
    }, [setColorScheme])

    return { colorScheme, switchTheme, metrics: monitor.current.reportMetrics() }
  }

    }, [])

      useNativeDriver: !screenReaderEnabled, // Disable for screen readers
      skipAnimation: prefersReducedMotion || screenReaderEnabled,
    }), [prefersReducedMotion, screenReaderEnabled])

    return { animationConfig, prefersReducedMotion, screenReaderEnabled }

  6. Memory Management

  Theme Asset Cleanup

  // utils/themeAssetManager.ts
  class ThemeAssetManager {
    private loadedAssets: Map<string, any> = new Map()
    private assetRefs: Map<string, number> = new Map()

    loadAsset(key: string, loader: () => Promise<any>) {
      const refCount = this.assetRefs.get(key) || 0
      this.assetRefs.set(key, refCount + 1)

      if (this.loadedAssets.has(key)) {
        return Promise.resolve(this.loadedAssets.get(key))
      }

      return loader().then(asset => {
        this.loadedAssets.set(key, asset)
        return asset
      })
    }

    releaseAsset(key: string) {
      const refCount = this.assetRefs.get(key) || 0
      if (refCount <= 1) {
        this.loadedAssets.delete(key)
        this.assetRefs.delete(key)
        // Trigger garbage collection hint
        if (global.gc) global.gc()
      } else {
        this.assetRefs.set(key, refCount - 1)
      }
    }

    clearUnusedAssets() {
      this.assetRefs.forEach((count, key) => {
        if (count === 0) {
          this.loadedAssets.delete(key)
          this.assetRefs.delete(key)
        }
      })
    }
  }

  interface TripsState {
    trips: Trip[]
    currentTrip: Trip | null
    isLoading: boolean
    error: string | null

    // Actions
    setTrips: (trips: Trip[]) => void
    addTrip: (trip: Trip) => void
    updateTrip: (id: string, updates: Partial<Trip>) => void
    deleteTrip: (id: string) => void
    setCurrentTrip: (trip: Trip | null) => void
    markForSync: (id: string) => void

  export const useTripsStore = create<TripsState>()(
    persist(
      immer((set) => ({
        trips: [],
        currentTrip: null,
        isLoading: false,
        error: null,

        setTrips: (trips) =>
          set((state) => {
            state.trips = trips
          }),

        addTrip: (trip) =>
          set((state) => {
            state.trips.push({
              ...trip,
              offline: true,
              syncStatus: 'pending',
            })
          }),

        updateTrip: (id, updates) =>
          set((state) => {
            const index = state.trips.findIndex((t) => t.id === id)
            if (index !== -1) {
              state.trips[index] = {
                ...state.trips[index],
                ...updates,
                syncStatus: 'pending',
              }
            }
          }),

        deleteTrip: (id) =>
          set((state) => {
            state.trips = state.trips.filter((t) => t.id !== id)
          }),

        setCurrentTrip: (trip) =>
          set((state) => {
            state.currentTrip = trip
          }),

        markForSync: (id) =>
          set((state) => {
            const trip = state.trips.find((t) => t.id === id)
            if (trip) {
              trip.syncStatus = 'pending'
            }
          }),
      })),
      {
        name: 'trips-storage',
        storage: createJSONStorage(() => zustandStorage),
        partialize: (state) => ({
          trips: state.trips,
          currentTrip: state.currentTrip,
        }),

  // Base API Service Class
  export abstract class BaseAPIService {
    protected tableName: string

    constructor(tableName: string) {
      this.tableName = tableName
    }

    // Check network connectivity
    protected async isOnline(): Promise<boolean> {
      const netInfo = await NetInfo.fetch()
      return netInfo.isConnected ?? false
    }

    // Queue operation for offline execution
    protected queueOfflineOperation(operation: any) {
      const { addToQueue } = useOfflineStore.getState()
      addToQueue({
        id: `${Date.now()}-${Math.random()}`,
        type: operation.type,
        payload: operation.payload,
        timestamp: Date.now(),
        retryCount: 0,
        status: 'pending',
      })
    }

    // Generic error handler
    protected handleError(error: any, context: string) {
      console.error(`[${context}] Error:`, error)

      // Check if it's a network error
      if (error.message?.includes('network') || error.code === 'NETWORK_ERROR') {
        return {
          error: 'Network error. Operation saved for offline sync.',
          offline: true,
        }
      }

      // Check for auth errors
      if (error.status === 401) {
        // Trigger re-authentication
        return {
          error: 'Authentication required',
          requiresAuth: true,
        }
      }

      return {
        error: error.message || 'An unexpected error occurred',
        details: error,
      }
    }
  }

  // services/api/trips.service.ts
  import { BaseAPIService } from './base.service'
  import { supabase } from './base.service'
  import { Trip, TripCreate, TripUpdate } from '@/types/api.types'
  import { useTripsStore } from '@/store/trips.store'

  class TripsService extends BaseAPIService {
    constructor() {
      super('trips')
    }

    async getTrips(): Promise<{ data?: Trip[], error?: any }> {
      try {
        // Check if online
        const online = await this.isOnline()

        if (!online) {
          // Return cached data
          const cachedTrips = useTripsStore.getState().trips
          return { data: cachedTrips }
        }

        // Fetch from Supabase
        const { data, error } = await supabase
          .from(this.tableName)
          .select(`
            *,
            participants:trip_participants(
              user:users(id, name, avatar_url)
            ),
            expenses:trip_expenses(
              id, amount, currency, category
            )
          `)
          .order('created_at', { ascending: false })

        if (error) throw error

        // Update local cache
        useTripsStore.getState().setTrips(data)

        return { data }
      } catch (error) {
        return { error: this.handleError(error, 'getTrips') }
      }
    }

    async createTrip(trip: TripCreate): Promise<{ data?: Trip, error?: any }> {
      try {
        const online = await this.isOnline()

        if (!online) {
          // Queue for offline sync
          const offlineTrip = {
            ...trip,
            id: `offline-${Date.now()}`,
            offline: true,
            syncStatus: 'pending',
          }

          this.queueOfflineOperation({
            type: 'CREATE_TRIP',
            payload: trip,
          })

          // Add to local store
          useTripsStore.getState().addTrip(offlineTrip as Trip)

          return { data: offlineTrip as Trip }
        }

        // Create in Supabase
        const { data, error } = await supabase
          .from(this.tableName)
          .insert(trip)
          .select()
          .single()

        if (error) throw error

        // Add to local store
        useTripsStore.getState().addTrip(data)

        return { data }
      } catch (error) {
        return { error: this.handleError(error, 'createTrip') }
      }
    }

    async updateTrip(id: string, updates: TripUpdate): Promise<{ data?: Trip, error?: any }> {
      try {
        const online = await this.isOnline()

        if (!online) {
          // Queue for offline sync
          this.queueOfflineOperation({
            type: 'UPDATE_TRIP',
            payload: { id, updates },
          })

          // Update local store
          useTripsStore.getState().updateTrip(id, {
            ...updates,
            syncStatus: 'pending',
          })

          return { data: { id, ...updates } as Trip }
        }

        // Update in Supabase
        const { data, error } = await supabase
          .from(this.tableName)
          .update(updates)
          .eq('id', id)
          .select()
          .single()

        if (error) throw error

        // Update local store
        useTripsStore.getState().updateTrip(id, data)

        return { data }
      } catch (error) {
        return { error: this.handleError(error, 'updateTrip') }
      }
    }

    async deleteTrip(id: string): Promise<{ error?: any }> {
      try {
        const online = await this.isOnline()

        if (!online) {
          // Queue for offline sync
          this.queueOfflineOperation({
            type: 'DELETE_TRIP',
            payload: { id },
          })

          // Mark as deleted in local store
          useTripsStore.getState().updateTrip(id, {
            deleted: true,
            syncStatus: 'pending',
          })

          return {}
        }

        // Delete from Supabase
        const { error } = await supabase
          .from(this.tableName)
          .delete()
          .eq('id', id)

        if (error) throw error

        // Remove from local store
        useTripsStore.getState().deleteTrip(id)

        return {}
      } catch (error) {
        return { error: this.handleError(error, 'deleteTrip') }
      }
    }

    // Real-time subscription
    subscribeToTrips(userId: string) {
      return supabase
        .channel('trips-channel')
        .on(
          'postgres_changes',
          {
            event: '*',
            schema: 'public',
            table: this.tableName,
            filter: `user_id=eq.${userId}`,
          },
          (payload) => {
            console.log('Real-time update:', payload)

            if (payload.eventType === 'INSERT') {
              useTripsStore.getState().addTrip(payload.new as Trip)
            } else if (payload.eventType === 'UPDATE') {
              useTripsStore.getState().updateTrip(payload.new.id, payload.new)
            } else if (payload.eventType === 'DELETE') {
              useTripsStore.getState().deleteTrip(payload.old.id)
            }
          }
        )
        .subscribe()
    }
  }

  export const tripsService = new TripsService()

  // hooks/api/useTrips.ts
  import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query'
  import { tripsService } from '@/services/api/trips.service'
  import { TripCreate, TripUpdate } from '@/types/api.types'
  import { useEffect } from 'react'
  import { useAuth } from './useAuth'

  export function useTrips() {
    const queryClient = useQueryClient()
    const { user } = useAuth()

    // Fetch trips
    const query = useQuery({
      queryKey: ['trips'],
      queryFn: () => tripsService.getTrips(),
      staleTime: 5 * 60 * 1000, // 5 minutes
      gcTime: 10 * 60 * 1000, // 10 minutes (formerly cacheTime)
      retry: (failureCount, error: any) => {
        // Don't retry if offline
        if (error?.offline) return false
        return failureCount < 3
      },
    })

    // Subscribe to real-time updates
    useEffect(() => {
      if (!user?.id) return

      const subscription = tripsService.subscribeToTrips(user.id)

      return () => {
        subscription.unsubscribe()
      }
    }, [user?.id])

    // Create trip mutation
    const createMutation = useMutation({
      mutationFn: (trip: TripCreate) => tripsService.createTrip(trip),
      onSuccess: (result) => {
        if (!result.error) {
          queryClient.invalidateQueries({ queryKey: ['trips'] })
        }
      },
      onError: (error) => {
        console.error('Create trip error:', error)
      },
    })

    // Update trip mutation
    const updateMutation = useMutation({
      mutationFn: ({ id, updates }: { id: string; updates: TripUpdate }) =>
        tripsService.updateTrip(id, updates),
      onSuccess: (result) => {
        if (!result.error) {
          queryClient.invalidateQueries({ queryKey: ['trips'] })
        }
      },
    })

    // Delete trip mutation
    const deleteMutation = useMutation({
      mutationFn: (id: string) => tripsService.deleteTrip(id),
      onSuccess: (result) => {
        if (!result.error) {
          queryClient.invalidateQueries({ queryKey: ['trips'] })
        }
      },
    })

    return {
      trips: query.data?.data || [],
      isLoading: query.isLoading,
      error: query.error || query.data?.error,
      refetch: query.refetch,
      createTrip: createMutation.mutate,
      updateTrip: updateMutation.mutate,
      deleteTrip: deleteMutation.mutate,
      isCreating: createMutation.isPending,
      isUpdating: updateMutation.isPending,
      isDeleting: deleteMutation.isPending,
    }
  }

  Offline Sync Service

  // services/offline/sync.service.ts
  import { useOfflineStore } from '@/store/offline.store'
  import { tripsService } from '@/services/api/trips.service'
  import NetInfo from '@react-native-community/netinfo'

  export class SyncService {
    private syncInProgress = false
    private syncInterval: NodeJS.Timeout | null = null

    // Start monitoring for sync opportunities
    startSyncMonitoring() {
      // Listen for network changes
      NetInfo.addEventListener((state) => {
        if (state.isConnected && !this.syncInProgress) {
          this.syncPendingOperations()
        }
      })

      // Check every 30 seconds
      this.syncInterval = setInterval(() => {
        this.syncPendingOperations()
      }, 30000)
    }

    stopSyncMonitoring() {
      if (this.syncInterval) {
        clearInterval(this.syncInterval)
        this.syncInterval = null
      }
    }

    async syncPendingOperations() {
      if (this.syncInProgress) return

      const netInfo = await NetInfo.fetch()
      if (!netInfo.isConnected) return

      const { queue, updateQueueItem, removeFromQueue } = useOfflineStore.getState()
      const pendingOps = queue.filter(op => op.status === 'pending')

      if (pendingOps.length === 0) return

      this.syncInProgress = true
      console.log(`Starting sync of ${pendingOps.length} operations`)

      for (const operation of pendingOps) {
        try {
          updateQueueItem(operation.id, { status: 'syncing' })

          let result
          switch (operation.type) {
            case 'CREATE_TRIP':
              result = await tripsService.createTrip(operation.payload)
              break
            case 'UPDATE_TRIP':
              result = await tripsService.updateTrip(
                operation.payload.id,
                operation.payload.updates
              )
              break
            case 'DELETE_TRIP':
              result = await tripsService.deleteTrip(operation.payload.id)
              break
            // Add more operation types as needed
          }

          if (result?.error) {
            throw result.error
          }

          // Success - remove from queue
          removeFromQueue(operation.id)
          console.log(`Synced operation: ${operation.type}`)
        } catch (error) {
          console.error(`Failed to sync operation:`, error)

          // Update retry count
          const newRetryCount = (operation.retryCount || 0) + 1

          if (newRetryCount >= 3) {
            // Max retries reached
            updateQueueItem(operation.id, {
              status: 'failed',
              error: error.message,
            })
          } else {
            // Will retry later
            updateQueueItem(operation.id, {
              status: 'pending',
              retryCount: newRetryCount,
            })
          }
        }
      }

      this.syncInProgress = false
      console.log('Sync completed')
    }
  }

  export function getErrorMessage(error: any): string {
    if (error instanceof APIError) {
      return errorMessages[error.code] || error.message
    }

    if (error?.offline) {
      return errorMessages.NETWORK_ERROR
    }

    if (error?.requiresAuth) {
      return errorMessages.AUTH_ERROR
    }

    if (error?.status === 404) {
      return errorMessages.NOT_FOUND
    }

    if (error?.status === 403) {
      return errorMessages.PERMISSION_DENIED
    }

    if (error?.status >= 500) {
      return errorMessages.SERVER_ERROR
    }

    return error?.message || errorMessages.SERVER_ERROR

  Chain of Thought: Why This API Architecture?

  Step 1: Problem Analysis
  - Mobile apps need to work offline (airports, foreign countries, poor connectivity)
  - Multiple users need real-time collaboration on trips
  - Data consistency between local and server state is critical

  // utils/requestDeduplicator.ts
  class RequestDeduplicator {
    private inFlightRequests = new Map<string, Promise<any>>()

    async deduplicate<T>(
      key: string,
      requestFn: () => Promise<T>
    ): Promise<T> {
      // Check if request is already in flight
      if (this.inFlightRequests.has(key)) {
        return this.inFlightRequests.get(key)!
      }

      // Start new request
      const promise = requestFn().finally(() => {
        this.inFlightRequests.delete(key)
      })

      this.inFlightRequests.set(key, promise)
      return promise
    }
  }

  Issue 3: No Rate Limiting

  Current Problem: Could overwhelm server with sync operations

  Refinement:
  // utils/rateLimiter.ts
  class RateLimiter {
    private queue: (() => Promise<any>)[] = []
    private processing = false
    private requestsPerSecond = 10

    async throttle<T>(fn: () => Promise<T>): Promise<T> {
      return new Promise((resolve, reject) => {
        this.queue.push(async () => {
          try {
            const result = await fn()
            resolve(result)
          } catch (error) {
            reject(error)
          }
        })

        if (!this.processing) {
          this.processQueue()
        }
      })
    }

    private async processQueue() {
      this.processing = true

      while (this.queue.length > 0) {
        const batch = this.queue.splice(0, this.requestsPerSecond)
        await Promise.all(batch.map(fn => fn()))
        await new Promise(resolve => setTimeout(resolve, 1000))
      }

      this.processing = false
    }
  }

  Issue 4: No Exponential Backoff

  Current Problem: Fixed retry intervals waste resources

  Refinement:
  // utils/retryWithBackoff.ts
  async function retryWithBackoff<T>(
    fn: () => Promise<T>,
    maxRetries = 3,
    baseDelay = 1000
  ): Promise<T> {
    let lastError: any

    for (let i = 0; i < maxRetries; i++) {
      try {
        return await fn()
      } catch (error) {
        lastError = error

        // Don't retry on client errors (4xx)
        if (error.status >= 400 && error.status < 500) {
          throw error
        }

        // Exponential backoff with jitter
        const delay = baseDelay * Math.pow(2, i) + Math.random() * 1000
        await new Promise(resolve => setTimeout(resolve, delay))
      }
    }

    throw lastError
  }

  // services/api/metrics.service.ts
  class APIMetrics {
    private metrics: Map<string, any[]> = new Map()

    recordAPICall(
      endpoint: string,
      duration: number,
      success: boolean,
      offline: boolean
    ) {
      const key = `${endpoint}-${success ? 'success' : 'failure'}`
      const current = this.metrics.get(key) || []

      current.push({
        duration,
        timestamp: Date.now(),
        offline,
      })

      this.metrics.set(key, current)

      // Send to analytics if online
      if (!offline) {
        analytics.track('api_call', {
          endpoint,
          duration,
          success,
        })
      }
    }

    getMetrics(endpoint: string) {
      const success = this.metrics.get(`${endpoint}-success`) || []
      const failure = this.metrics.get(`${endpoint}-failure`) || []

      return {
        totalCalls: success.length + failure.length,
        successRate: success.length / (success.length + failure.length),
        avgDuration: success.reduce((a, b) => a + b.duration, 0) / success.length,
        p95Duration: this.percentile(success.map(s => s.duration), 0.95),
      }
    }

    private percentile(arr: number[], p: number) {
      const sorted = arr.sort((a, b) => a - b)
      const index = Math.ceil(sorted.length * p) - 1
      return sorted[index]
    }
  }

      try {
        const result = await retryWithBackoff(() =>
          this.rateLimiter.throttle(() =>
            this.deduplicator.deduplicate(operation, fn)

        )

        this.metrics.recordAPICall(
          operation,
          Date.now() - start,
          true,
          !online
        )

        return result
      } catch (error) {
        this.metrics.recordAPICall(
          operation,
          Date.now() - start,
          false,
          !online
        )
        throw error
      }
    }
