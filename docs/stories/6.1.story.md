# Story 6.1: Real-time Updates (MVP)

## Status
Draft

## Story
**As a** trip member,  
**I want** to see changes made by others in real-time,  
**So that** everyone stays synchronized on trip expenses

## Acceptance Criteria
1. Subscribe to trip changes on mount
2. Update UI when expenses added
3. Update UI when expenses deleted
4. Update member list on join/leave
5. Show toast for remote changes
6. Unsubscribe on unmount
7. Handle connection errors gracefully

### Realtime Setup
```typescript
// Hook for trip realtime
export const useTripRealtime = (tripId: string) => {
  const queryClient = useQueryClient();
  const { user } = useAuth();
  
  useEffect(() => {
    // Subscribe to expenses
    const expenseChannel = supabase
      .channel(`trip-${tripId}-expenses`)
      .on(
        'postgres_changes',
        {
          event: '*',
          schema: 'public',
          table: 'expenses',
          filter: `trip_id=eq.${tripId}`,
        },
        (payload) => {
          // Skip if own change
          if (payload.new?.created_by === user?.id) return;
          
          // Update cache
          queryClient.invalidateQueries(['expenses', tripId]);
          
          // Show toast
          if (payload.eventType === 'INSERT') {
            showToast(`New expense added: ${payload.new.description}`);
          }
        }
      )
      .subscribe();
      
    // Subscribe to members
    const memberChannel = supabase
      .channel(`trip-${tripId}-members`)
      .on(
        'postgres_changes',
        {
          event: '*',
          schema: 'public',
          table: 'trip_members',
          filter: `trip_id=eq.${tripId}`,
        },
        (payload) => {
          // Update cache
          queryClient.invalidateQueries(['trip', tripId]);
          
          if (payload.eventType === 'INSERT') {
            showToast('New member joined the trip');
          }
        }
      )
      .subscribe();
    
    // Cleanup
    return () => {
      supabase.removeChannel(expenseChannel);
      supabase.removeChannel(memberChannel);
    };
  }, [tripId, user?.id]);
};
```
