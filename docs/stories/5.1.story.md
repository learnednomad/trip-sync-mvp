### Form Structure
```typescript
interface ExpenseForm {
  amount: number;
  description: string;
  paidBy: string; // user ID
  participants: string[]; // user IDs
}

// Equal split calculation
const calculateSplits = (amount: number, participants: string[]) => {
  const splitAmount = Math.round(amount / participants.length);
  return participants.map(userId => ({
    user_id: userId,
    amount_owed: splitAmount,
  }));
};
```

### Save Implementation
```typescript
const saveExpense = async (form: ExpenseForm, tripId: string) => {
  const expense = {
    trip_id: tripId,
    amount: form.amount,
    description: form.description,
    paid_by: form.paidBy,
    created_by: currentUser.id,
  };
  
  const isOnline = useNetworkStatus();
  
  if (!isOnline) {
    // Queue for sync
    await offlineQueue.add({
      type: 'CREATE_EXPENSE',
      payload: { expense, participants: form.participants },
    });
    return;
  }
  
  // Online save
  const { data, error } = await supabase
    .from('expenses')
    .insert(expense)
    .select()
    .single();
    
  // Create splits
  const splits = calculateSplits(form.amount, form.participants);
  await supabase
    .from('expense_splits')
    .insert(splits.map(s => ({ ...s, expense_id: data.id })));
};
```
