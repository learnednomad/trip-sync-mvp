### Cache Structure
```typescript
// Cache keys
const CACHE_KEYS = {
  trips: 'cache:trips',
  expenses: (tripId: string) => `cache:expenses:${tripId}`,
  user: 'cache:user',
  timestamp: (key: string) => `${key}:timestamp`,
};

// Cache wrapper
export const cache = {
  set: async <T>(key: string, data: T) => {
    try {
      storage.set(key, JSON.stringify(data));
      storage.set(CACHE_KEYS.timestamp(key), Date.now());
    } catch (error) {
      console.error('Cache write failed:', error);
    }
  },
  
  get: async <T>(key: string): Promise<T | null> => {
    try {
      const timestamp = storage.getNumber(CACHE_KEYS.timestamp(key));
      
      // Check expiry (30 days)
      if (timestamp && Date.now() - timestamp > 30 * 24 * 60 * 60 * 1000) {
        storage.delete(key);
        return null;
      }
      
      const data = storage.getString(key);
      return data ? JSON.parse(data) : null;
    } catch {
      return null;
    }
  },
  
  clear: () => {
    const keys = storage.getAllKeys();
    keys.filter(k => k.startsWith('cache:')).forEach(k => {
      storage.delete(k);
    });
  },
};
```

### Integration with Queries
```typescript
// In useTrips hook
const cachedTrips = await cache.get<Trip[]>(CACHE_KEYS.trips);

return useQuery({
  queryKey: ['trips'],
  queryFn: async () => {
    const trips = await fetchTrips();
    cache.set(CACHE_KEYS.trips, trips);
    return trips;
  },
  initialData: cachedTrips,
  staleTime: 5 * 60 * 1000,
});
```
