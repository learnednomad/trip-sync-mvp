### Update Profile
```typescript
const updateProfile = async (name: string) => {
  const { error } = await supabase
    .from('users')
    .update({ name })
    .eq('id', user.id);
    
  if (!error) {
    // Update local state
    authStore.updateUser({ name });
    showToast('Profile updated!');
  }
};
```
