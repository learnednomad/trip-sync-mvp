### Join Trip Flow
```typescript
const joinTrip = async (inviteCode: string) => {
  // Normalize code
  const code = inviteCode.toUpperCase().trim();
  
  // Validate format
  if (!/^[A-Z0-9]{6}$/.test(code)) {
    throw new Error('Invalid code format');
  }
  
  // Call RPC
  const { data, error } = await supabase
    .rpc('join_trip', { 
      invite_code: code 
    });
    
  if (error) {
    if (error.code === '23505') {
      throw new Error('You are already a member');
    }
    throw error;
  }
  
  return data;
};
```

### Database Function
```sql
-- This should exist in Supabase
CREATE OR REPLACE FUNCTION join_trip(invite_code TEXT)
RETURNS trips AS $$
DECLARE
  trip_record trips;
BEGIN
  -- Find trip
  SELECT * INTO trip_record
  FROM trips
  WHERE trips.invite_code = join_trip.invite_code;
  
  -- Insert member
  INSERT INTO trip_members (trip_id, user_id, role)
  VALUES (trip_record.id, auth.uid(), 'member');
  
  RETURN trip_record;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;
```
