### Balance Calculation
```typescript
interface UserBalance {
  userId: string;
  userName: string;
  totalPaid: number;
  totalOwed: number;
  netBalance: number; // positive = owed money, negative = owes money
}

const calculateBalances = (expenses: Expense[], members: TripMember[]) => {
  const balances: Record<string, UserBalance> = {};
  
  // Initialize for all members
  members.forEach(member => {
    balances[member.user_id] = {
      userId: member.user_id,
      userName: member.users.name,
      totalPaid: 0,
      totalOwed: 0,
      netBalance: 0,
    };
  });
  
  // Calculate totals
  expenses.forEach(expense => {
    // Add to paid total
    balances[expense.paid_by].totalPaid += expense.amount;
    
    // Add to owed totals
    expense.expense_splits.forEach(split => {
      balances[split.user_id].totalOwed += split.amount_owed;
    });
  });
  
  // Calculate net balances
  Object.values(balances).forEach(balance => {
    balance.netBalance = balance.totalPaid - balance.totalOwed;
  });
  
  return Object.values(balances);
};

// Simplified settlements (who pays whom)
const calculateSettlements = (balances: UserBalance[]) => {
  const creditors = balances.filter(b => b.netBalance > 0);
  const debtors = balances.filter(b => b.netBalance < 0);
  const settlements: Settlement[] = [];
  
  // Simple algorithm - not optimal but works for MVP
  creditors.forEach(creditor => {
    let remaining = creditor.netBalance;
    
    debtors.forEach(debtor => {
      if (remaining > 0 && debtor.netBalance < 0) {
        const amount = Math.min(remaining, Math.abs(debtor.netBalance));
        settlements.push({
          from: debtor.userName,
          to: creditor.userName,
          amount: amount,
        });
        remaining -= amount;
        debtor.netBalance += amount;
      }
    });
  });
  
  return settlements;
};
```
