### Trip Creation Flow
```typescript
const createTrip = async (data: TripForm) => {
  // Generate invite code
  const inviteCode = generateInviteCode(); // 6 chars
  
  // Insert trip
  const { data: trip, error } = await supabase
    .from('trips')
    .insert({
      name: data.name,
      start_date: data.startDate,
      end_date: data.endDate,
      created_by: user.id,
      invite_code: inviteCode,
    })
    .select()
    .single();
    
  // Add creator as member
  await supabase
    .from('trip_members')
    .insert({
      trip_id: trip.id,
      user_id: user.id,
      role: 'owner',
    });
    
  return trip;
};
```

### Date Handling
- Use native date pickers
- Store as ISO strings
- Default to null if not set
- Validate end > start
