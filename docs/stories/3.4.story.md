### Data Fetching
```typescript
const useTripDetail = (tripId: string) => {
  return useQuery({
    queryKey: ['trip', tripId],
    queryFn: async () => {
      const { data, error } = await supabase
        .from('trips')
        .select(`
          *,
          trip_members (
            user_id,
            role,
            users (
              id,
              name,
              email
            )
          )
        `)
        .eq('id', tripId)
        .single();
        
      if (error) throw error;
      return data;
    },
  });
};
```
