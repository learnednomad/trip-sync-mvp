# Story 4.3: Offline Queue (MVP)

## Status
Draft

## Story
**As a** user,  
**I want** my offline changes to sync when I'm back online,  
**So that** I don't lose any data I created while offline

## Acceptance Criteria
1. Queue mutations when offline
2. Store queue in MMKV
3. Process queue when online
4. Retry failed items (max 3)
5. Clear items after success
6. Show sync status indicator
7. Handle conflicts (last-write-wins)

### Queue Structure
```typescript
interface QueuedAction {
  id: string;
  type: 'CREATE_TRIP' | 'CREATE_EXPENSE' | 'DELETE_EXPENSE';
  payload: any;
  timestamp: number;
  retryCount: number;
  userId: string;
}

// Queue implementation
export const offlineQueue = {
  add: async (action: Omit<QueuedAction, 'id' | 'timestamp' | 'retryCount'>) => {
    const item: QueuedAction = {
      ...action,
      id: generateId(),
      timestamp: Date.now(),
      retryCount: 0,
    };
    
    const queue = await getQueue();
    queue.push(item);
    await saveQueue(queue);
  },
  
  process: async () => {
    const queue = await getQueue();
    const pending = [...queue];
    
    for (const action of pending) {
      try {
        await processAction(action);
        // Remove from queue
        await removeFromQueue(action.id);
      } catch (error) {
        action.retryCount++;
        if (action.retryCount >= 3) {
          // Give up, remove from queue
          await removeFromQueue(action.id);
        } else {
          // Update retry count
          await updateInQueue(action);
        }
      }
    }
  },
};

// Process specific actions
const processAction = async (action: QueuedAction) => {
  switch (action.type) {
    case 'CREATE_TRIP':
      return await supabase.from('trips').insert(action.payload);
    case 'CREATE_EXPENSE':
      return await supabase.from('expenses').insert(action.payload);
    case 'DELETE_EXPENSE':
      return await supabase.from('expenses').delete().eq('id', action.payload.id);
  }
};
```
