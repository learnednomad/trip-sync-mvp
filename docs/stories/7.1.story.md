### Unit Test Examples
```typescript
// Balance calculation tests
describe('calculateBalances', () => {
  it('should calculate correct net balances', () => {
    const expenses = [
      {
        id: '1',
        amount: 100,
        paid_by: 'user1',
        expense_splits: [
          { user_id: 'user1', amount_owed: 33.33 },
          { user_id: 'user2', amount_owed: 33.33 },
          { user_id: 'user3', amount_owed: 33.34 },
        ],
      },
    ];
    
    const balances = calculateBalances(expenses, members);
    
    expect(balances.user1.netBalance).toBe(66.67); // Paid 100, owes 33.33
    expect(balances.user2.netBalance).toBe(-33.33);
    expect(balances.user3.netBalance).toBe(-33.34);
  });
  
  it('should handle empty expenses', () => {
    const balances = calculateBalances([], members);
    expect(balances.every(b => b.netBalance === 0)).toBe(true);
  });
});

// Offline queue tests
describe('offlineQueue', () => {
  beforeEach(() => {
    storage.clearAll();
  });
  
  it('should queue actions when offline', async () => {
    const action = {
      type: 'CREATE_EXPENSE',
      payload: { amount: 50 },
      userId: 'user1',
    };
    
    await offlineQueue.add(action);
    
    const queue = await offlineQueue.getAll();
    expect(queue).toHaveLength(1);
    expect(queue[0].type).toBe('CREATE_EXPENSE');
  });
});
```

### Integration Test Example
```typescript
// Offline sync test
it('should sync queued items when online', async () => {
  // Add item while offline
  NetInfo.mock({ isConnected: false });
  await createExpense(mockExpense);
  
  // Verify queued
  const queue = await offlineQueue.getAll();
  expect(queue).toHaveLength(1);
  
  // Go online and sync
  NetInfo.mock({ isConnected: true });
  await offlineQueue.process();
  
  // Verify synced
  const newQueue = await offlineQueue.getAll();
  expect(newQueue).toHaveLength(0);
  
  // Verify in database
  const { data } = await supabase
    .from('expenses')
    .select()
    .eq('description', mockExpense.description);
  expect(data).toHaveLength(1);
});
```
