### Query Implementation
```typescript
const useTrips = () => {
  return useQuery({
    queryKey: ['trips'],
    queryFn: async () => {
      const { data, error } = await supabase
        .from('trips')
        .select(`
          *,
          trip_members(count)
        `)
        .order('created_at', { ascending: false });
        
      if (error) throw error;
      return data;
    },
  });
};
```
